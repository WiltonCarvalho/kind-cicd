git clone --depth 1 https://github.com/OpenVPN/easy-rsa.git
cp -r easy-rsa/easyrsa3 mtls

############################
cd mtls

./easyrsa init-pki

./easyrsa build-ca nopass
openssl pkcs12 -export -nokeys -in pki/ca.crt -out ca.p12 -legacy -password pass:changeit

############################
./easyrsa build-server-full server nopass
openssl pkcs12 -export -inkey pki/private/server.key -in pki/issued/server.crt -out server.p12 -name server -legacy -password pass:changeit

############################
cd mtls

./easyrsa build-client-full client1 nopass
openssl pkcs12 -export -inkey pki/private/client1.key -in pki/issued/client1.crt -out client1.p12 -name client1 -legacy -password pass:changeit

############################
./easyrsa revoke client1
./easyrsa gen-crl
chmod +r pki/crl.pem

############################

chmod +Xr ../mtls -R
